一开始使能接收中断，在接受中断中将数据存入环形buffer

编写思路：
1.
定义一个全局变量g_RecvChar，用于存放接收到的1个字节
定义一个函数StartUART1Recv在main函数中调用，用于使能接收中断
static uint8_t g_RecvChar;
void StartUART1Recv(void)
{
	HAL_UART_Receive_IT(&huart1, &g_RecvChar, 1);
}

//注：HAL_UART_Receive_IT这个函数内部有一句 __HAL_UART_ENABLE_IT(huart, UART_IT_RXNE);用于使能RXNE中断
2.
定义一个环形缓冲区 g_RecvBuf[100]用于存放接收到的数据
使用定义过的环形缓冲区类型来定义一个结构体 g_uart1_rx_bufs
接收中断回调函数HAL_UART_RxCpltCallback中将收到的1个字节数据存入环形缓冲区

static uint8_t g_RecvBuf[100];
static circle_buf g_uart1_rx_bufs;
void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
	g_rx_cplt = 1;   //上一节用于修改状态接收状态
	circle_buf_write(&g_uart1_rx_bufs, g_RecvChar);
}

3.
定义一个读取数据的函数UART1GetChar
在该函数中调用前几节定义的读取环形缓冲区的函数来读接收到的存于环形缓冲区的数据
int UART1GetChar(uint8_t *pVal)
{
	return circle_buf_read(&g_uart1_rx_bufs, pVal);
}

4.
接收一个字节数据后UART中断服务函数HAL_UART_IRQHandler会读取DR寄存器，自动清除了RXNE标志位，
因此需要重新使能RXNE中断，在接收中断回调函数中添加一行：
void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
	g_rx_cplt = 1;	
	circle_buf_write(&g_uart1_rx_bufs, g_RecvChar);
	HAL_UART_Receive_IT(&huart1, &g_RecvChar, 1);	
}

环形缓冲区需要初始化，StartUART1Recv函数中加一行：
void StartUART1Recv(void)
{
	circle_buf_init(&g_uart1_rx_bufs, 100, g_RecvBuf);
	HAL_UART_Receive_IT(&huart1, &g_RecvChar, 1);
}


·整体思路：
StartUART1Recv函数初始化环形缓冲区结构体，使能RXNE中断，
HAL_UART_RxCpltCallback接收中断回调函数将接收到的数据存入环形缓冲区，
UART1GetChar函数读取数据

编写的串口程序：

static volatile int g_tx_cplt = 0;
static volatile int g_rx_cplt = 0;
static uint8_t g_RecvChar;
static uint8_t g_RecvBuf[100];
static circle_buf g_uart1_rx_bufs;

void HAL_UART_TxCpltCallback(UART_HandleTypeDef *huart)
{
	g_tx_cplt = 1;
}

void Wait_Tx_Complete(void)
{
	while (g_tx_cplt == 0);
	g_tx_cplt = 0;
}

void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
	g_rx_cplt = 1;
	
	/* put data to circle buffer */
	circle_buf_write(&g_uart1_rx_bufs, g_RecvChar);
	
	/* re-enable rxne interrupt */
	HAL_UART_Receive_IT(&huart1, &g_RecvChar, 1);	
}
void Wait_Rx_Complete(void)
{
	while (g_rx_cplt == 0);
	g_rx_cplt = 0;
}

void StartUART1Recv(void)
{
	/* init circle buffer */
	circle_buf_init(&g_uart1_rx_bufs, 100, g_RecvBuf);
	
	/* enable rxne interrupt */
	HAL_UART_Receive_IT(&huart1, &g_RecvChar, 1);
}

int UART1GetChar(uint8_t *pVal)
{
	return circle_buf_read(&g_uart1_rx_bufs, pVal);
}



编写的环形缓冲区程序：
typedef struct circle_buf {
	uint32_t r;
	uint32_t w;
	uint32_t len;
	uint8_t *buf;
}circle_buf, *p_circle_buf;

void circle_buf_init(p_circle_buf pCircleBuf, uint32_t len, uint8_t *buf);

int circle_buf_read(p_circle_buf pCircleBuf, uint8_t *pVal);

int circle_buf_write(p_circle_buf pCircleBuf, uint8_t val);



void circle_buf_init(p_circle_buf pCircleBuf, uint32_t len, uint8_t *buf)
{
	pCircleBuf->r = pCircleBuf->w = 0;
	pCircleBuf->len = len;
	pCircleBuf->buf = buf;
}

int circle_buf_read(p_circle_buf pCircleBuf, uint8_t *pVal)
{
	if (pCircleBuf->r != pCircleBuf->w)
	{
		*pVal = pCircleBuf->buf[pCircleBuf->r];
		
		pCircleBuf->r++;
		
		if (pCircleBuf->r == pCircleBuf->len)
			pCircleBuf->r = 0;
		return 0;
	}
	else
	{
		return -1;
	}
}

int circle_buf_write(p_circle_buf pCircleBuf, uint8_t val)
{
	uint32_t next_w;
	
	next_w = pCircleBuf->w + 1;
	if (next_w == pCircleBuf->len)
		next_w = 0;
	
	if (next_w != pCircleBuf->r)
	{
		pCircleBuf->buf[pCircleBuf->w] = val;
		pCircleBuf->w = next_w;
		return 0;
	}
	else
	{
		return -1;
	}
}
