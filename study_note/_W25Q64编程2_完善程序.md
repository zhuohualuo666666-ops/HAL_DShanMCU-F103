1.回调函数系列：
static volatile int g_spi1_tx_complete = 0;
static volatile int g_spi1_rx_complete = 0;
static volatile int g_spi1_txrx_complete = 0;

void HAL_SPI_TxCpltCallback(SPI_HandleTypeDef *hspi)
{
	if (hspi == &hspi1)
	{
		g_spi1_tx_complete = 1;
	}
}

void Wait_SPI1_TxCplt(int timeout)
{
	while (g_spi1_tx_complete == 0 && timeout--)
	{
		HAL_Delay(1);
	}
	g_spi1_tx_complete = 0;
}

void HAL_SPI_RxCpltCallback(SPI_HandleTypeDef *hspi)
{
	if (hspi == &hspi1)
	{
		g_spi1_rx_complete = 1;
	}
}

void Wait_SPI1_RxCplt(int timeout)
{
	while (g_spi1_rx_complete == 0 && timeout--)
	{
		HAL_Delay(1);
	}
	g_spi1_rx_complete = 0;
}

void HAL_SPI_TxRxCpltCallback(SPI_HandleTypeDef *hspi)
{
	if (hspi == &hspi1)
	{
		g_spi1_txrx_complete = 1;
	}
}

void Wait_SPI1_TxRxCplt(int timeout)
{
	while (g_spi1_txrx_complete == 0 && timeout--)
	{
		HAL_Delay(1);
	}
	g_spi1_txrx_complete = 0;
}

思路：
中断调用回调函数时将状态修改，跳出完成标志函数的阻塞，
防止死循环引入超时时间，超时时间可自己设置

2.内部函数与外部函数：
#define SPI_FLASH_TIMEOUT 1000     //超时时间设为1000ms

void Wait_SPI1_TxCplt(int timeout);
void Wait_SPI1_TxRxCplt(int timeout);
void Wait_SPI1_RxCplt(int timeout);


extern SPI_HandleTypeDef hspi1;

/* 内部函数  */
static void SPIFlash_Select(void)
{
	HAL_GPIO_WritePin(GPIOB, GPIO_PIN_9, GPIO_PIN_RESET);
}

static void SPIFlash_DeSelect(void)
{
	HAL_GPIO_WritePin(GPIOB, GPIO_PIN_9, GPIO_PIN_SET);
}

static int SPIFlash_WriteEnable(void)
{
	uint8_t buf[1] = {0x06}; //声明要写使能
	SPIFlash_Select();       //片选
	HAL_SPI_Transmit_IT(&hspi1, buf, 1); //发送数据0x06
	Wait_SPI1_TxCplt(SPI_FLASH_TIMEOUT); //等待发送完成调用回调函数
	SPIFlash_DeSelect();            //取消片选
}

static int SPIFlash_ReadStatus(void)
{
	uint8_t txbuf[2] = {0x05, 0xff};  //0X05表示读状态寄存器1指令，0xff仅用于提供时钟，无实际意义
	uint8_t rxbuf[2] = {0, 0};
	
	SPIFlash_Select();  //片选
	HAL_SPI_TransmitReceive_IT(&hspi1, txbuf, rxbuf, 2);//发送数据到rxbuf中
	Wait_SPI1_TxRxCplt(SPI_FLASH_TIMEOUT);//等待发送完成调用回调函数
	SPIFlash_DeSelect();        //取消片选
	return rxbuf[1];        //读出rxbuf[1]中存储的W25Q64返回的状态寄存器值
}

static int SPIFlash_WaitReady(void)
{
	while (SPIFlash_ReadStatus() & 1 == 1);  //等待烧写擦除操作完成，烧写擦除操作进行时bit0即BUSY位为1
}

/* 对外的函数 */

int SPIFlash_ReadID(void)
{
	uint8_t txbuf[2] = {0x9F, 0xff};//0X05表示读ID指令，0xff仅用于提供时钟，无实际意义
	uint8_t rxbuf[2] = {0, 0};
	
	SPIFlash_Select();           //片选
	HAL_SPI_TransmitReceive_IT(&hspi1, txbuf, rxbuf, 2);/发送数据到rxbuf中
	Wait_SPI1_TxRxCplt(SPI_FLASH_TIMEOUT);//等待发送完成调用回调函数
	SPIFlash_DeSelect();  //取消片选
	return rxbuf[1];   //读出rxbuf[1]中存储的W25Q64返回的状态寄存器值
}

int SPIFlash_EraseSector(uint32_t addr)
{
	uint8_t txbuf[4] = {0x20}; //0x20表示擦除命令
	
	/* write enable */
	SPIFlash_WriteEnable();
	
	/* erase */
	txbuf[1]= (addr>>16) & 0xff;  //扇区擦除命令（0x20）后面需要跟一个 24 位的扇区地址，SPI是高位字节先传输，取出原地址23~16位
	txbuf[2]= (addr>>8) & 0xff;   //取出源地址第15~8位
	txbuf[3]= (addr>>0) & 0xff;    //取出源地址第7~0位

	SPIFlash_Select();
	HAL_SPI_Transmit_IT(&hspi1, txbuf, 4);
	Wait_SPI1_TxCplt(SPI_FLASH_TIMEOUT);
	SPIFlash_DeSelect();   //同上
	
	/* wait ready */
	SPIFlash_WaitReady();
	
	return 0;
}

int SPIFlash_Write(uint32_t addr, uint8_t *datas, uint32_t len)
{
	uint8_t txbuf[4] = {0x02};//0x02表示页编程命令
	
	/* write enable */
	SPIFlash_WriteEnable();
	
	/* program */
	txbuf[1]= (addr>>16) & 0xff;
	txbuf[2]= (addr>>8) & 0xff;
	txbuf[3]= (addr>>0) & 0xff;

	SPIFlash_Select();
	
	/* 发送命令和地址 */
	HAL_SPI_Transmit_IT(&hspi1, txbuf, 4);
	Wait_SPI1_TxCplt(SPI_FLASH_TIMEOUT);
	
	/* 发送数据 */
	HAL_SPI_Transmit_IT(&hspi1, datas, len);
	Wait_SPI1_TxCplt(SPI_FLASH_TIMEOUT);
	
	SPIFlash_DeSelect();
	
	/* wait ready */
	SPIFlash_WaitReady();
	
	return 0;
}

int SPIFlash_Read(uint32_t addr, uint8_t *datas, uint32_t len)
{
	uint8_t txbuf[4] = {0x03};//0x03表示读数据命令
		
	/* read */
	txbuf[1]= (addr>>16) & 0xff;
	txbuf[2]= (addr>>8) & 0xff;
	txbuf[3]= (addr>>0) & 0xff;

	SPIFlash_Select();
	
	/* 发送命令和地址 */
	HAL_SPI_Transmit_IT(&hspi1, txbuf, 4);
	Wait_SPI1_TxCplt(SPI_FLASH_TIMEOUT);
	
	/* 读取数据 */
	HAL_SPI_Receive_IT(&hspi1, datas, len);
	Wait_SPI1_RxCplt(SPI_FLASH_TIMEOUT);
	
	SPIFlash_DeSelect();
		
	return 0;
}
