<!-- 环形缓冲区源码解析 -->
typedef struct circle_buf {
    uint32_t r;
    uint32_t w;
    uint32_t len;
    uint8_t *buf;
}circle_buf, *p_circle_buf;

//定义一个结构体类型circle_buf，给它的指针类型起名p_circle_buf，里面包含读写位置指针，环形缓冲区长度，指向环形缓冲区数据的指针

void circle_buf_init(p_circle_buf pCircleBuf, uint32_t len, uint8_t *buf) //pCircleBuf是指向结构体变量的指针，定义结构体变量时&后填入这里；定义一个足够大的数组放在在buf这里
{
    pCircleBuf->r = pCircleBuf->w = 0;
    pCircleBuf->len = len;
    pCircleBuf->buf = buf;   //环形缓冲区相当于一维数组，放在这里就是地址，把它赋给指向环形缓冲区的指针，将结构体与环形缓冲区联系起来
}

int circle_buf_read(p_circle_buf pCircleBuf, uint8_t *pVal) //pCircleBuf是指向结构体变量的指针，pVal是要读取的数据
{
    if (pCircleBuf->r != pCircleBuf->w) //表示非空
    {
        *pVal = pCircleBuf->buf[pCircleBuf->r];  //把环形缓冲区里读指针指向的数据取出来
        
        pCircleBuf->r++; //读指针移动
        
        if (pCircleBuf->r == pCircleBuf->len)  //读指针溢出回到初始位置
            pCircleBuf->r = 0;
        return 0;
    }
    else
    {
        return -1; //表示空
    }
}

int circle_buf_write(p_circle_buf pCircleBuf, uint8_t val)//pCircleBuf是指向结构体变量的指针，Val是要写入的数据
{
    uint32_t next_w;  //下一个写位置，用于区分空与满
    
    next_w = pCircleBuf->w + 1;
    if (next_w == pCircleBuf->len)   //溢出
        next_w = 0;
    
    if (next_w != pCircleBuf->r) //表示还有空位置
    {
        pCircleBuf->buf[pCircleBuf->w] = val;  //把要写的数据val放到环形缓冲区里
        pCircleBuf->w = next_w;     //写指针移动
        return 0;
    }
    else
    {
        return -1;//表示满
    }
}
